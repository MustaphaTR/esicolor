ZII_task_force_ship_organization = {
	name = BUL_varna_naval_dockyard_organization
	include = esicolor_generic_task_force_ship_organization
	# icon = GFX_idea_generic_naval_manufacturer_2

	allowed = {
		has_dlc = "Arms Against Tyranny"
	}
	visible = {
		owner = {
			NOT = { has_country_flag = pacific_fleet_naval_manufacturer_disabled }
			OR = {
				original_tag = ZII
				AND = {
					has_global_flag = custom_designers_only
					owns_state = 211
				}
			}
		}
	}
	available = {
		if = {
			limit = { has_global_flag = custom_designers_only }

			owner = {
				211 = { state_designer_availabile = yes }
			}
		}
	}

	add_trait = { 
		token = BUL_mio_trait_an_aegean_sea_fleet
		name = BUL_an_aegean_sea_fleet
		icon = GFX_generic_mio_department_icon_facilities
		special_trait_background = yes

		position = { x=9 y=4 }

		visible = { has_global_flag = custom_designer_traits_enabled }
		available = {
			owner = {
				any_controlled_state = {
					BUL_state_is_in_the_aegean_sea = yes
					dockyard > 0
				}
			}
		}

		organization_modifier = {
			military_industrial_organization_research_bonus = 0.1
		}
		production_bonus = {
			production_capacity_factor = 0.1
		}
	}
}
