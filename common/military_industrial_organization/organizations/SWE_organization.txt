	include = esicolor_generic_mobile_tank_organization
	icon = GFX_idea_SWE_landsverk

	allowed = {
		has_dlc = "Arms Against Tyranny"
	}
	visible = {
		owner = {
			NOT = { has_country_flag = fast_tank_manufacturer_disabled }
			OR = {
				original_tag = IEB
				AND = {
					has_global_flag = custom_designers_only
					owns_state = 138
				}
			}
		}
	}
	available = {
		if = {
			limit = { has_global_flag = custom_designers_only }

			owner = {
				138 = { state_designer_availabile = yes }
			}
		}
	}

	trait = {
		token = SWE_mio_trait_german_landsverk_tech_exchange
		name = SWE_mio_trait_german_landsverk_tech_exchange
		icon = GFX_generic_mio_department_icon_facilities
		special_trait_background = yes

		position = { x=0 y=1 }
		relative_position_id = generic_mio_streamlined_production

		all_parents = { generic_mio_streamlined_production }
		visible = { has_global_flag = custom_designer_traits_enabled }
		available = {
			if = {
				limit = { has_global_flag = custom_designers_only_core }

				52 = {
					OR = {
						is_core_of = FROM
						AND = {
							is_core_of = owner
							owner = { is_subject_of = FROM }
						}
					}
				}
			}
			else = {
				52 = {
					OR = {
						is_owned_by = FROM
						owner = { is_subject_of = FROM }
					}
				}
			}
		}

		production_bonus = {
			production_capacity_factor = 0.03
		}
		organization_modifier = {
			military_industrial_organization_research_bonus = 0.08
		
		}
	}
}
