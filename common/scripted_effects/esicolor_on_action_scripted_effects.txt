recalculate_recruitment_costs = {
	set_variable = {
		THIS.general_recruitment_cost = {
			value = 5
			multiply = THIS.army_leaders^num
			multiply = {
				value = 1
				add = THIS.modifier@army_leader_cost_factor
			}
			min = 200
		}
	}
	set_variable = {
		THIS.general_recruitment_cost_minus = {
			value = THIS.general_recruitment_cost
			multiply = -1
		}
	}

	set_variable = {
		THIS.marshal_recruitment_cost = {
			value = THIS.general_recruitment_cost

		}

			value = global.claiming_cost?20

				value = THIS.claim_timer_@PREV
				divide = global.claiming_length?30
			}
		}
	}
	add_political_power = var:claim_refund

	add_to_variable = { THIS.dynamic_pp_gain_modifier = 0.1 }
	clamp_variable = {
		var = THIS.dynamic_pp_gain_modifier
		max = 0
		min = -2000
	}
	remove_from_array = {
		array = THIS.states_being_claimed
		value = PREV
	}
}
claim_action_effect = {
	ROOT = { add_threat = 1 }

	add_claim_by = ROOT
	owner = { set_temp_variable = { state_owner = THIS } }

	if = {
		limit = {
			var:state_owner = {
				NOT = {
					tag = ROOT
					is_in_faction_with = ROOT
				}
			}
		}

		ROOT = {
			if = {
				limit = {
					OR = {
						has_global_flag = claiming_give_war_goals
						has_global_flag = claiming_declare_war
					}
				}

				create_wargoal = {
					type = annex_everything
					target = state_owner
				}
			}
			if = {
				limit = {
					has_global_flag = claiming_declare_war
					NOT = { has_war_with = state_owner }
				}

				declare_war_on = {
					type = annex_everything
					target = state_owner
				}
			}
		}
	}
}

	set_variable = { temp_trait^3 = 0 }
	set_variable = { temp_trait^4 = 0 }

	set_variable = { temp_trait^9 = 0 }
	set_variable = { temp_trait_index = 0 }

}
esicolor_unit_leader_traits_tt = {
	if = {
		limit = { # double-checks so it doesn't run too many times
			check_variable = { current_step < num_traits }
			check_variable = { current_step < 11 }
		}

		esicolor_meta_madness = yes
		add_to_variable = { current_step = 1 }
		esicolor_unit_leader_traits_tt = yes
	}
}
esicolor_meta_madness = {
	if = {
		limit = { check_variable = { temp_index < global.traits_array^num } } # double-checks so it doesn't run too many times

		meta_effect = {
			text = {
				if = {
					limit = { has_trait = [TRAIT] }

					set_variable = { temp_trait^temp_trait_index = token:[TRAIT] }
					add_to_variable = { temp_trait_index = 1 }
				}
			}
			TRAIT = "[?global.traits_array^temp_index.GetTokenKey]"
		}
		add_to_variable = { temp_index = 1 }
		esicolor_meta_madness = yes # effect runs itself until it reaches end of the array
	}
}
