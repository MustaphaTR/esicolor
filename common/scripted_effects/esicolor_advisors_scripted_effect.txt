generate_generic_popular_figurehead = {
	meta_effect = {
		text = {
			generate_character = {
				token_base = [TAG]_ba_popular_figurehead
				portraits = { army = { small = GFX_idea_unknown_advisor } }
			}

			add_advisor_role = {
				character = [TAG]_ba_popular_figurehead

				advisor = {
					slot = political_advisor
					traits = { popular_figurehead }

					ai_will_do = {
						factor = 1

						modifier = {
							has_stability > 0.35
							has_stability < 0.5

							add = 9
						}
					}

		text = {
			generate_character = {
				token_base = [TAG]_df_air_strategic_bombing
			}

			add_advisor_role = {
				character = [TAG]_df_air_strategic_bombing

				advisor = {
					slot = high_command
					ledger = air
					cost = 100
					traits = { air_strategic_bombing_2 }
				}
			}
		}
		TAG = "[THIS.GetTag]"
	}
}
retire_air_strategic_bombing = {
	random_character = {
		limit = {
			is_advisor = yes

				has_trait = air_strategic_bombing_1
				has_trait = air_strategic_bombing_2
				has_trait = air_strategic_bombing_3
			}
		}

		retire = yes
	}
}
