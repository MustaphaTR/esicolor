	esicolor_foreign_generals_gui = {
		context_type = decision_category
		window_name = "esicolor_foreign_generals_gui_window"

		properties = {
			esicolor_foreign_generals_gui_priority = {
				frame = esicolor_foreign_generals_gui_priority_frame
			}
		}

		effects = {
			esicolor_foreign_generals_gui_priority_click = {
				if = {
					limit = { has_country_flag = foreign_generals_named_only }

					clr_country_flag = foreign_generals_named_only
					set_country_flag = foreign_generals_named_prioritised
					clr_country_flag = foreign_generals_no_priority
					set_variable = { esicolor_foreign_generals_gui_priority_frame = 1 }
				}
				else_if = {
					limit = { has_country_flag = foreign_generals_named_prioritised }

					clr_country_flag = foreign_generals_named_only
					clr_country_flag = foreign_generals_named_prioritised
					set_country_flag = foreign_generals_no_priority
					set_variable = { esicolor_foreign_generals_gui_priority_frame = 3 }
				}
				else = {
					set_country_flag = foreign_generals_named_only
					clr_country_flag = foreign_generals_named_prioritised
					clr_country_flag = foreign_generals_no_priority
					set_variable = { esicolor_foreign_generals_gui_priority_frame = 2 }
				}
			}
			esicolor_foreign_generals_gui_priority_right_click = {
				if = {
					limit = { has_country_flag = foreign_generals_no_priority }

					clr_country_flag = foreign_generals_named_only
					set_country_flag = foreign_generals_named_prioritised
					clr_country_flag = foreign_generals_no_priority
					set_variable = { esicolor_foreign_generals_gui_priority_frame = 1 }
				}
				else_if = {
					limit = { has_country_flag = foreign_generals_named_only }

					clr_country_flag = foreign_generals_named_only
					clr_country_flag = foreign_generals_named_prioritised
					set_country_flag = foreign_generals_no_priority
					set_variable = { esicolor_foreign_generals_gui_priority_frame = 3 }
				}
				else = {
					set_country_flag = foreign_generals_named_only
					clr_country_flag = foreign_generals_named_prioritised
					clr_country_flag = foreign_generals_no_priority
					set_variable = { esicolor_foreign_generals_gui_priority_frame = 2 }
				}
			}
		}
	}
