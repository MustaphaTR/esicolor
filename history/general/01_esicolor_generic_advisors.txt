# For minor countries where advisors are not defined
# Generate them from a generic template

# Ideology Advisors - available field don't work with add_advisor_role, needed for Horn of Africa focus checks
every_possible_country = {
	limit = {
		NOT = { original_tag = OOO }
		has_custom_staunch_monarchist = no
	}

	meta_effect = {
		text = {
			generate_character = {
				token_base = [TAG]_a_staunch_monarchist
				advisor = {
					slot = political_advisor
					traits = { staunch_monarchist }

					available = {
						if = {
							limit = { has_focus_tree = horn_of_africa_tree }

							OR = {
								has_government = neutrality
								has_completed_focus = HOA_a_change_in_course
							}
						}
					}

					ai_will_do = {
						factor = 0

						modifier = {
							has_country_flag = ai_go_neutrality
							NOT = { has_government = neutrality }

							add = 30
						}
					}
				}
				portraits = { army = { small = GFX_idea_unknown_advisor } }
			}
		}
		TAG = "[THIS.GetTag]"
	}
}
